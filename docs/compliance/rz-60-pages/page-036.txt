def _export_founder_only() -> bool:
    raw = getattr(settings, "EXPORT_PLATFORM_FOUNDER_ONLY", True)
    if isinstance(raw, str):
        return raw.lower() in ("1", "true", "yes")
    return bool(raw)

        "export_kind": export_kind,
