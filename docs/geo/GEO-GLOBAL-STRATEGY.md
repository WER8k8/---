## 🛠️ 技术实施文件

### 1. 多语言Schema组件
```vue
<!-- components/business/MultiLangSchema.vue -->
<template>
  <div>
    <SchemaMarkup
      v-for="locale in ['zh', 'en']"
      :key="locale"
      :type="schemaType"
      :data="getLocalizedData(locale)"
    />
  </div>
</template>
```

### 2. 语言自动切换
```typescript
// composables/useI18nSEO.ts
export function useI18nSEO() {
  const { locale } = useI18n()

  function setLocalizedSEO(product: Product) {
    const isZh = locale.value === 'zh'

    setSEO({
      title: isZh ? product.name : product.nameEn,
      description: isZh ? product.description : product.descriptionEn,
      keywords: isZh ? product.keywords : product.keywordsEn
    })
  }

  return { setLocalizedSEO }
}
```
